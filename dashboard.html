Quick Start — SDK
// C# — Arena System SDK
using ModzSquad;

var client = new ModzClient("zs_live_xK9mP3q...");

var result = await client.Auth.LoginAsync(new LoginRequest {
    LicenseKey = "𝐆𝐳𝐞𝐧𝐨𝐌𝐨𝐝𝐳-AB45F-REH54-XK92M-7PL3N",
    Hwid       = HwidHelper.Get(),
    Version    = "1.0.0"
});

if (result.Success)
    Console.WriteLine($"Welcome {result.User.Username}!");
# Python — Arena System SDK
import modz_squad as zs

client = zs.ModzClient("zs_live_xK9mP3q...")

result = client.auth.login(
    license_key="𝐆𝐳𝐞𝐧𝐨𝐌𝐨𝐝𝐳-AB45F-REH54-XK92M-7PL3N",
    hwid=zs.get_hwid(),
    version="1.0.0"
)

if result["success"]:
    print(f"Welcome {result['user']['username']}!")
// Node.js — Arena System SDK
const { ModzClient } = require('modz-squad');

const client = new ModzClient('zs_live_xK9mP3q...');

const result = await client.auth.login({
  licenseKey: '𝐆𝐳𝐞𝐧𝐨𝐌𝐨𝐝𝐳-AB45F-REH54-XK92M-7PL3N',
  hwid:       getHwid(),
  version:    '1.0.0'
});

if (result.success) console.log(`Welcome ${result.user.username}!`);
